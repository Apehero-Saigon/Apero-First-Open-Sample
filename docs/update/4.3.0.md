# Integration Guide — v4.3.0

> **Upgrade target:** v4.1.8 → v4.3.0

---

## 1. Pick a flow: D0 or D7

The SDK ships two first-open flows. Extend exactly **one** of the two abstract
activities.

|                 | **D0 (`AstronexFirstOpenActivity`)**                                                           | **D7 (`AstronexD7FirstOpenActivity`)**                                                              |
|-----------------|------------------------------------------------------------------------------------------------|-----------------------------------------------------------------------------------------------------|
| Onboard         | Up to 3 native onboard pages + welcome (+ welcome dup) + fullscreen flex + invisible inventory | 3 onboard pages: page 1 = native, page 2 = banner, page 3 = banner; optional fullscreen+fullscreen2 |
| Welcome screen  | Yes (`welcomeUiConfig`)                                                                        | No                                                                                                  |
| Required ad IDs | `FOD0AdsConfig`                                                                                | `FOD7AdsConfig`                                                                                     |

---

## 2. Implement your `FirstOpenActivity`

Each flow has a dedicated `data class` that lists exactly the ad IDs it
consumes. Every ID is a **required** constructor parameter — a missing field
is a compile error, not a silent `null`. The `setUpAdConfig()` abstract method
ties the type system to the flow you chose.

### 2.1 D0 — `AstronexFirstOpenActivity`

```kotlin
class FirstOpenActivity : AstronexFirstOpenActivity() {

    override fun onProvideAppInfo(): AppInfo = AppInfo(
        packageName = BuildConfig.APPLICATION_ID,
        versionCode = BuildConfig.VERSION_CODE,
        versionName = BuildConfig.VERSION_NAME,
    )

    override fun setUpAdConfig(): FOD0AdsConfig = FOD0AdsConfig(
        interSplashHighId = BuildConfig.inter_splash_high,
        interSplashId = BuildConfig.inter_splash,
        bannerSplashId = BuildConfig.banner_splash,
        nativeLanguageHighId = BuildConfig.native_language_high,
        nativeLanguageId = BuildConfig.native_language,
        nativeLanguageDupHighId = BuildConfig.native_language_dup_high,
        nativeLanguageDupId = BuildConfig.native_language_dup,
        nativeWelcomeHighId = BuildConfig.native_welcome_high,
        nativeWelcomeId = BuildConfig.native_welcome,
        nativeWelcomeDupHighId = BuildConfig.native_welcome_dup_high,
        nativeWelcomeDupId = BuildConfig.native_welcome_dup,
        nativeOnboard1HighId = BuildConfig.native_onboard_1_high,
        nativeOnboard1Id = BuildConfig.native_onboard_1,
        nativeOnboard2HighId = BuildConfig.native_onboard_2_high,
        nativeOnboard2Id = BuildConfig.native_onboard_2,
        nativeOnboard3HighId = BuildConfig.native_onboard_3_high,
        nativeOnboard3Id = BuildConfig.native_onboard_3,
        nativeOnboardFullscreenHighId = BuildConfig.native_ob_fullscr_high,
        nativeOnboardFullscreenId = BuildConfig.native_ob_fullscr,
        nativeFullscrFlexHighId = BuildConfig.native_ob_fullscr_high,
        // loadAdTimeOut defaults to 30000L; override only if you need a different timeout
    )

    override fun onSetupFOConfig(): FOConfig =
        FOConfig.Builder()
            .setPrimaryColor(Color.Magenta)
            .setCallback(callback)
            .setSplashUiConfig(splashConfig)
            .setLanguageUiConfig(languageConfig)
            .setOnboardUiConfig(onboardConfig)
            .setWelcomeUiConfig(welcomeUiConfig)
            .setPrepareUiConfig(prepareUiConfig)
            .build()
}
```

### 2.2 D7 — `AstronexD7FirstOpenActivity`

```kotlin
class FirstOpenActivity : AstronexD7FirstOpenActivity() {

        nativeLanguageId = BuildConfig.native_language,
        nativeLanguageDupHighId = BuildConfig.native_language_dup_high,
        nativeLanguageDupId = BuildConfig.native_language_dup,
        nativeOnboard1HighId = BuildConfig.native_onboard_1_high,
        nativeOnboard1Id = BuildConfig.native_onboard_1,
        nativeOnboardFullscreenHighId = BuildConfig.native_ob_fullscr_high,
        nativeOnboardFullscreenId = BuildConfig.native_ob_fullscr,
        nativeOnboardFullscreen2HighId = BuildConfig.native_ob_fullscr_2_high,
        nativeOnboardFullscreen2Id = BuildConfig.native_ob_fullscr_2,
    )

    override fun onSetupFOConfig(): FOConfig =
        FOConfig.Builder()
            .setPrimaryColor(Color.Magenta)
            .setCallback(callback)
            .setSplashUiConfig(splashConfig)
            .setLanguageUiConfig(languageConfig)
            .setOnboardUiConfig(onboardConfig)
            .setPrepareUiConfig(prepareUiConfig)

```

> **Notes**
> - Returning `FOD7AdsConfig` from an `AstronexFirstOpenActivity` (or vice-versa) is a compile-time
    error.
> - Forgetting any ad-ID parameter is a compile-time error (`No value passed for parameter 'X'`).
> - Pass `null` explicitly when you genuinely don't have an ID for that slot.

## 3. Read Firebase Remote Config — `FOCallback.onRemoteConfigResult`

```kotlin
import com.astronex.firstopen.app.FOCallback
import com.astronex.firstopen.data.repository.RemoteConfigRepository

private val callback = object : FOCallback() {
    override suspend fun onRemoteConfigResult(remoteConfigRepository: RemoteConfigRepository) {
        val flag = remoteConfigRepository.getBooleanConfig("my_flag", false)
        val text = remoteConfigRepository.getStringConfig("my_text", "")
        val count = remoteConfigRepository.getIntConfig("my_count", 0)
        val price = remoteConfigRepository.getLongConfig("my_price", 0L)
        val ratio = remoteConfigRepository.getFloatConfig("my_ratio", 0f)
    }
}
```

Wire it via `.setCallback(callback)` inside `onSetupFOConfig()` (see §2.1 / §2.2).

> **Notes**
> - Fires once per app process, only on a successful fetch+activate (skipped on failure / timeout).
> - Invoked on a background dispatcher — switch to the main thread before touching UI state.
> - Use the typed suspend getters above; they read from the SDK's internal `DataStore`.
